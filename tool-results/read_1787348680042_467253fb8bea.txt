   558→   558→-- ═══════════════════════════════════════════════════════════════
   559→   559→
   560→   560→SELECT '--- Users Created ---' AS info;
   561→   561→SELECT email, name, role, is_active, email_verified
   562→   562→FROM public.profiles
   563→   563→ORDER BY role, email;
   564→   564→
   565→   565→SELECT '--- Jobs ---' AS info;
   566→   566→SELECT j.title, p.name AS employer, j.location, j.salary_min, j.salary_max, j.job_type, j.sector, j.status
   567→   567→FROM public.jobs j JOIN public.profiles p ON p.id = j.employer_id
   568→   568→ORDER BY j.created_at;
   569→   569→
   570→   570→SELECT '--- Applications ---' AS info;
   571→   571→SELECT j.title AS job, c.name AS candidate, a.status, a.created_at::date AS applied_on
   572→   572→FROM public.applications a
   573→   573→JOIN public.jobs j ON j.id = a.job_id
   574→   574→JOIN public.profiles c ON c.id = a.candidate_id
   575→   575→ORDER BY a.created_at;
   576→   576→
   577→   577→SELECT '--- Interviews ---' AS info;
   578→   578→SELECT c.name AS candidate, j.title AS job, i.scheduled_at, i.duration_min || ' min' AS duration, i.status, i.location
   579→   579→FROM public.interviews i
   580→   580→JOIN public.applications a ON a.id = i.application_id
   581→   581→JOIN public.jobs j ON j.id = a.job_id
   582→   582→JOIN public.profiles c ON c.id = a.candidate_id
   583→   583→ORDER BY i.scheduled_at;
   584→   584→
   585→   585→SELECT '--- Totals ---' AS info;
   586→   586→SELECT
   587→   587→  (SELECT count(*) FROM public.profiles) AS total_users,
   588→   588→  (SELECT count(*) FROM public.jobs) AS total_jobs,
   589→   589→  (SELECT count(*) FROM public.applications) AS total_applications,
   590→   590→  (SELECT count(*) FROM public.interviews) AS total_interviews,
   591→   591→  (SELECT count(*) FROM public.housing_requests) AS total_housing_requests,
   592→   592→  (SELECT count(*) FROM public.payments) AS total_payments,
   593→   593→  (SELECT count(*) FROM public.messages) AS total_messages,
   594→   594→  (SELECT count(*) FROM public.notifications) AS total_notifications,
   595→   595→  (SELECT count(*) FROM public.cvs) AS total_cvs,
   596→   596→  (SELECT count(*) FROM public.references) AS total_references,
   597→   597→  (SELECT count(*) FROM public.documents) AS total_documents;
