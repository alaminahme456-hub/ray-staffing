   244→
   245→  const filteredJobs = useMemo(() => {
   246→    let jobs = [...allJobs]
   247→
   248→    if (keyword.trim()) {
   249→      const kw = keyword.toLowerCase()
   250→      jobs = jobs.filter(
   251→        (j) =>
   252→          j.title.toLowerCase().includes(kw) ||
   253→          j.employer.toLowerCase().includes(kw) ||
   254→          j.description.toLowerCase().includes(kw)
   255→      )
   256→    }
   257→    if (location.trim()) {
   258→      const loc = location.toLowerCase()
   259→      jobs = jobs.filter((j) => j.location.toLowerCase().includes(loc))
   260→    }
   261→    if (fIndustry !== 'all') jobs = jobs.filter((j) => j.industry === fIndustry)
   262→    if (fType !== 'all') jobs = jobs.filter((j) => j.type === fType)
   263→    if (fExperience !== 'all') jobs = jobs.filter((j) => j.experience === fExperience)
   264→    if (fMode !== 'all') jobs = jobs.filter((j) => j.mode === fMode)
   265→    if (fSalaryMin) {
   266→      const min = parseInt(fSalaryMin.replace(/[^0-9]/g, ''))
   267→      if (!isNaN(min)) {
   268→        jobs = jobs.filter((j) => {
   269→          const match = j.salary.match(/£([0-9,]+)/)
   270→          return match ? parseInt(match[1].replace(',', '')) >= min : true
   271→        })
   272→      }
   273→    }
   274→    if (fSalaryMax) {
   275→      const max = parseInt(fSalaryMax.replace(/[^0-9]/g, ''))
   276→      if (!isNaN(max)) {
   277→        jobs = jobs.filter((j) => {
   278→          const matches = j.salary.match(/£([0-9,]+).*?£([0-9,]+)/)
   279→          if (matches) {
   280→            return parseInt(matches[2].replace(',', '')) <= max
   281→          }
   282→          return true
   283→        })
   284→      }
   285→    }
   286→    if (fDate !== 'all') {
   287→      const now = Date.now()

   289→      jobs = jobs.filter((j) => {
   290→        const days = { today: 1, week: 7, month: 30 }
   291→        const d = days[fDate as keyof typeof days] || 0
   292→        const postedDays = j.posted.match(/(\d+)/)
   293→        const pd = postedDays ? parseInt(postedDays[1]) : 1
   294→        const unit = j.posted.includes('week') ? 7 : 1
   295→        return pd * unit <= d
   296→      })
   297→    }
   298→
   299→    if (sort === 'recent') {
   300→      jobs.sort((a, b) => {
   301→        const da = a.posted.match(/(\d+)/)?.[1] || '0'
   302→        const db = b.posted.match(/(\d+)/)?.[1] || '0'
   303→        const ua = a.posted.includes('week') ? 7 : 1
   304→        const ub = b.posted.includes('week') ? 7 : 1
   305→        return parseInt(da) * ua - parseInt(db) * ub
   306→      })
   307→    } else if (sort === 'salary-high') {
   308→      jobs.sort((a, b) => {
   309→        const sa = parseInt((a.salary.match(/£([0-9,]+).*?£([0-9,]+)/)?.[2] || a.salary.match(/£([0-9,]+)/)?.[1] || '0').replace(',', ''))
   310→        const sb = parseInt((b.salary.match(/£([0-9,]+).*?£([0-9,]+)/)?.[2] || b.salary.match(/£([0-9,]+)/)?.[1] || '0').replace(',', ''))
   311→        return sb - sa
   312→      })
   313→    }
   314→
   315→    return jobs
