   168→/* ------------------------------------------------------------------ */
   169→/*  External store for jobs (avoids setState in effects)               */
   170→/* ------------------------------------------------------------------ */
   171→
   172→type JobsSnapshot = {
   173→  jobs: JobRow[]
   174→  appCounts: Record<string, number>
   175→  loading: boolean
   176→}
   177→
   178→let snapshot: JobsSnapshot = {
   179→  jobs: [],
   180→  appCounts: {},
   181→  loading: true,
   182→}
   183→
   184→const storeListeners = new Set<() => void>()
   185→
   186→function emitSnapshotChange() {
   187→  for (const fn of storeListeners) fn()
   188→}
   189→
   190→function subscribeToSnapshot(listener: () => void) {
   191→  storeListeners.add(listener)
   192→  return () => { storeListeners.delete(listener) }
   193→}
   194→
   195→function getSnapshot(): JobsSnapshot {
   196→  return snapshot
   197→}
   198→
   199→function getServerSnapshot(): JobsSnapshot {
   200→  return { jobs: [], appCounts: {}, loading: true }
   201→}
   202→
   203→/** Fetch jobs + app counts and push into the external store */
   204→async function fetchJobsIntoStore(employerId: string) {
   205→  snapshot = { ...snapshot, loading: true }
   206→  emitSnapshotChange()
   207→
   208→  const supabase = createClient()
   209→
   210→  const { data: jobsData, error: jobsErr } = await supabase
   211→    .from('jobs')
   212→    .select('*')
   213→    .eq('employer_id', employerId)
   214→    .order('created_at', { ascending: false })
   215→
   216→  if (jobsErr) {
   217→    snapshot = { ...snapshot, loading: false }
   218→    emitSnapshotChange()
   219→    return
   220→  }
   221→
   222→  const rows = (jobsData as JobRow[]) ?? []
   223→
   224→  if (rows.length === 0) {
   225→    snapshot = { jobs: [], appCounts: {}, loading: false }
   226→    emitSnapshotChange()
   227→    return
   228→  }
   229→
   230→  const jobIds = rows.map((j) => j.id)
   231→
   232→  const { data: appsData } = await supabase
   233→    .from('applications')
   234→    .select('job_id')
   235→    .in('job_id', jobIds)
   236→
   237→  const counts: Record<string, number> = {}
   238→  for (const app of appsData ?? []) {
   239→    counts[app.job_id] = (counts[app.job_id] || 0) + 1
   240→  }
   241→
   242→  snapshot = { jobs: rows, appCounts: counts, loading: false }
   243→  emitSnapshotChange()
   244→}

   250→export default function EmployerVacancies() {

   253→

   265→  const [editingJob, setEditingJob] = useState<JobRow | null>(null)

   267→  const [submitting, setSubmitting] = useState(false)
   268→
   269→  /* ================================================================ */
   270→  /*  Subscribe to external store & trigger initial fetch              */
   271→  /* ================================================================ */
   272→
   273→  useEffect(() => {
   274→    if (!user?.id) return
   275→    fetchJobsIntoStore(user.id)
   276→  }, [user])
   277→
   278→  /* ================================================================ */

   280→  /* ================================================================ */
   281→
   282→  const filtered = store.jobs.filter((v) => {
   283→    const q = search.toLowerCase()
   284→    const matchSearch =
   285→      v.title.toLowerCase().includes(q) ||
   286→      (v.location ?? '').toLowerCase().includes(q)
   287→    const matchStatus = statusFilter === 'all' || v.status === statusFilter
   288→    return matchSearch && matchStatus
   289→  })
   290→
   291→  /* ================================================================ */
   292→  /*  Dialog helpers                                                   */
   293→  /* ================================================================ */
   294→
   295→  function openCreateDialog() {
   296→    setEditingJob(null)
   297→    setForm(EMPTY_FORM)
   298→    setDialogOpen(true)
   299→  }
   300→
   301→  function openEditDialog(job: JobRow) {
   302→    setEditingJob(job)
   303→    setForm(jobToForm(job))
   304→    setDialogOpen(true)
   305→  }
   306→
   307→  function closeDialog() {
   308→    setDialogOpen(false)
   309→    setEditingJob(null)
   310→    setForm(EMPTY_FORM)
   311→  }
   312→
   313→  /* ================================================================ */
   314→  /*  Submit (create / update)                                         */
   315→  /* ================================================================ */
   316→
   317→  async function handleSubmit(status: 'draft' | 'active') {
   318→    if (!form.title.trim()) {
   319→      toast.error('Job title is required')
   320→      return
   321→    }
   322→    if (!form.listing.trim()) {
   323→      toast.error('Job description (listing) is required')
   324→      return
   325→    }
   326→
   327→    setSubmitting(true)
   328→    const supabase = createClient()
   329→
   330→    const payload = {
   331→      employer_id: user!.id,
   332→      title: form.title.trim(),
   333→      listing: form.listing.trim(),
   334→      requirements: form.requirements.trim() || null,
   335→      location: form.location.trim() || null,
   336→      salary_min: form.salary_min ? Number(form.salary_min) : null,
   337→      salary_max: form.salary_max ? Number(form.salary_max) : null,
   338→      job_type: form.job_type || null,
   339→      sector: form.sector.trim() || null,
   340→      description: form.listing.trim(),
   341→      status,
   342→    }
   343→
   344→    if (editingJob) {
   345→      const { error } = await supabase
   346→        .from('jobs')
   347→        .update(payload)
   348→        .eq('id', editingJob.id)
   349→
   350→      if (error) {
   351→        toast.error('Failed to update vacancy')
   352→        setSubmitting(false)
   353→        return
   354→      }
   355→      toast.success('Vacancy updated successfully')
   356→    } else {
   357→      const { error } = await supabase.from('jobs').insert(payload)
   358→
   359→      if (error) {
   360→        toast.error('Failed to create vacancy')
   361→        setSubmitting(false)
   362→        return
   363→      }
   364→      toast.success(
   365→        status === 'draft'
   366→          ? 'Vacancy saved as draft'
   367→          : 'Vacancy published successfully',
   368→      )
   369→    }
   370→
   371→    setSubmitting(false)
   372→    closeDialog()
   373→    if (user?.id) fetchJobsIntoStore(user.id)
   374→  }
   375→
   376→  /* ================================================================ */
   377→  /*  Status toggle                                                    */
   378→  /* ================================================================ */
   379→
   380→  async function updateStatus(jobId: string, newStatus: string) {
   381→    const supabase = createClient()
   382→    const { error } = await supabase
   383→      .from('jobs')
   384→      .update({ status: newStatus })
   385→      .eq('id', jobId)

   387→    if (error) {
   388→      toast.error(
   389→        `Failed to ${newStatus === 'active' ? 'reopen' : newStatus} vacancy`,
   390→      )
   391→      return
   392→    }
   393→
   394→    const label =
   395→      newStatus === 'active'
   396→        ? 'reopened'
   397→        : newStatus === 'paused'
   398→          ? 'paused'
   399→          : 'closed'
   400→    toast.success(`Vacancy ${label} successfully`)
   401→    if (user?.id) fetchJobsIntoStore(user.id)
   402→  }

   430→          <Plus className="w-4 h-4 mr-2" /> Create Vacancy

   443→          <Input
   444→            placeholder="Search by title or location…"
