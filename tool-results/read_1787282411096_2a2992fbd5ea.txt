    68→function formatDate(dateStr: string): string {
    69→  const d = new Date(dateStr)
    70→  return d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' })
    71→}
    72→
    73→function formatDateTime(dateStr: string): string {
    74→  const d = new Date(dateStr)
    75→  return d.toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' }) +
    76→    ' ' + d.toLocaleTimeString('en-GB', { hour: '2-digit', minute: '2-digit' })
    77→}
    78→
    79→function getInitials(name: string | null): string {
    80→  if (!name) return '??'
    81→  return name.split(' ').map(n => n[0]).join('').toUpperCase().slice(0, 2)
    82→}

   126→      setProfiles((data as Profile[]) ?? [])

   128→      const message = err instanceof Error ? err.message : 'Failed to load users'
   129→      toast.error(message)

   131→      setLoading(false)
   132→    }
   133→  }, [])
   134→
   135→  // Fetch profiles on initial mount via useState initializer
   136→  // to avoid useEffect + setState (React 19 strict lint).
   137→  const [fetched] = useState(() => {
   138→    // IIFE on initial render to trigger fetch without useEffect
   139→    createClient()
   140→      .from('profiles')
   141→      .select('*')
   142→      .order('created_at', { ascending: false })
   143→      .then(({ data, error }) => {
   144→        if (error) {
   145→          toast.error(error.message)
   146→        } else {
   147→          setProfiles((data as Profile[]) ?? [])
   148→        }
   149→        setLoading(false)

   152→  })
   153→
   154→  const filtered = profiles.filter((u) => {
   155→    const name = (u.name ?? '').toLowerCase()
   156→    const email = (u.email ?? '').toLowerCase()
   157→    const matchesSearch = name.includes(search.toLowerCase()) || email.includes(search.toLowerCase())
   158→    const matchesRole = roleFilter === 'ALL' || u.role === roleFilter
   159→    return matchesSearch && matchesRole
   160→  })
   161→

   163→    if (togglingId) return

   165→    const actionLabel = newStatus ? 'activated' : 'deactivated'
   166→    setTogglingId(profile.id)
   167→
   168→    try {
   169→      const supabase = createClient()

   171→        .from('profiles')
   172→        .update({ is_active: newStatus })
   173→        .eq('id', profile.id)
   174→
   175→      if (error) throw error
   176→
   177→      setProfiles(prev => prev.map(p =>
   178→        p.id === profile.id ? { ...p, is_active: newStatus } : p
   179→      ))
   180→
   181→      toast.success(`${profile.name ?? profile.email} has been ${actionLabel}`)
   182→      if (selectedUser?.id === profile.id) {
   183→        setSelectedUser(prev => prev ? { ...prev, is_active: newStatus } : null)
   184→      }
   185→    } catch (err: unknown) {
   186→      const message = err instanceof Error ? err.message : `Failed to ${actionLabel} user`
   187→      toast.error(message)
   188→    } finally {
   189→      setTogglingId(null)
   190→    }
   191→  }
   192→

   223→                <Users className="h-5 w-5 text-[#4ade80]" />

   236→                <UserCheck className="h-5 w-5 text-[#4ade80]" />
   237→              </div>
   238→              <div>
   239→                <p className="text-2xl font-bold text-[#FAF8F5]">
   240→                  {profiles.filter(p => p.is_active).length}
   241→                </p>

   279→      {/* Filters */}

   286→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-[#7a8f8a]" />

   302→                  ? 'bg-[#1a3a1e] text-[#4ade80] hover:bg-[#1f4a24] border-[#2a5e2e]'

   304→              }
   305→              onClick={() => setRoleFilter(role)}
   306→            >
   307→              {role === 'ALL' ? 'All Roles' : (roleLabels[role] ?? role)}
   308→            </Button>
   309→          ))}
   310→        </div>
   311→      </motion.div>
   312→
   313→      {/* Users Table */}
   314→      <motion.div initial={{ opacity: 0, y: 20 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: 0.1 }}>
   315→        <Card className="bg-[#0a1a0e] border-[#1a2e1e]">
   316→          <CardContent className="p-0">
   317→            <div className="overflow-x-auto max-h-[600px] overflow-y-auto">
   318→              <Table>

   320→                  <TableRow className="border-[#1a2e1e] bg-[#0f1f12] hover:bg-[#0f1f12]">

   322→                    <TableHead className="text-[#7a8f8a] font-medium">Email</TableHead>
   323→                    <TableHead className="text-[#7a8f8a] font-medium">Role</TableHead>
   324→                    <TableHead className="text-[#7a8f8a] font-medium">Status</TableHead>
   325→                    <TableHead className="text-[#7a8f8a] font-medium">Created</TableHead>
   326→                    <TableHead className="text-[#7a8f8a] font-medium text-right">Actions</TableHead>
   327→                  </TableRow>
   328→                </TableHeader>
   329→                <TableBody>
   330→                  {loading ? (
   331→                    <TableSkeleton />
   332→                  ) : filtered.length === 0 ? (
   333→                    <TableRow className="border-[#1a2e1e]">
   334→                      <TableCell colSpan={6} className="h-40 text-center">
   335→                        <div className="flex flex-col items-center gap-2 text-[#5a6f65]">
   336→                          <Users className="h-10 w-10" />
   337→                          <p className="text-sm">No users found</p>
   338→                          <p className="text-xs">Try adjusting your search or filter</p>
   339→                        </div>
   340→                      </TableCell>
   341→                    </TableRow>
   342→                  ) : (
   343→                    <AnimatePresence mode="popLayout">
   344→                      {filtered.map((user) => (
   345→                        <motion.tr
   346→                          key={user.id}
   347→                          layout
   348→                          initial={{ opacity: 0, y: 10 }}
   349→                          animate={{ opacity: 1, y: 0 }}
   350→                          exit={{ opacity: 0, y: -10 }}
   351→                          transition={{ duration: 0.2 }}
   352→                          className="border-[#1a2e1e] hover:bg-[#0f1f12] transition-colors"
   353→                        >
   354→                          <TableCell>
   355→                            <div className="flex items-center gap-3">
   356→                              <div className="h-8 w-8 rounded-full bg-[#1a3a1e] text-[#4ade80] flex items-center justify-center text-xs font-semibold shrink-0">
   357→                                {getInitials(user.name)}
   358→                              </div>
   359→                              <span className="font-medium text-[#FAF8F5] whitespace-nowrap">
   360→                                {user.name ?? 'Unnamed'}
   361→                              </span>
   362→                            </div>
   363→                          </TableCell>
   364→                          <TableCell className="text-[#7a8f8a] whitespace-nowrap">
   365→                            {user.email ?? '—'}
   366→                          </TableCell>

   370→                              className={roleColors[user.role ?? ''] ?? 'bg-gray-100 text-gray-500 border-gray-200'}

   388→                              </span>

   392→                            {formatDate(user.created_at)}

   396→                              <Button
   397→                                variant="ghost"
   398→                                size="icon"
   399→                                className="h-8 w-8 text-[#7a8f8a] hover:text-[#FAF8F5] hover:bg-[#1a2e1e]"
   400→                                onClick={() => setSelectedUser(user)}
   401→                              >
   402→                                <Eye className="h-4 w-4" />
   403→                              </Button>
   404→                              <DropdownMenu>
   405→                                <DropdownMenuTrigger asChild>
   406→                                  <Button
   407→                                    variant="ghost"
   408→                                    size="icon"
   409→                                    className="h-8 w-8 text-[#7a8f8a] hover:text-[#FAF8F5] hover:bg-[#1a2e1e]"
   410→                                  >

   449→
   450→      {/* Results count */}

   475→              <div className="flex items-center gap-4">
   476→                <div className="h-14 w-14 rounded-full bg-[#1a3a1e] text-[#4ade80] flex items-center justify-center text-lg font-semibold shrink-0">
